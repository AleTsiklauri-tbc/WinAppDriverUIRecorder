package tbc.automation.steps;

import io.appium.java_client.windows.WindowsDriver;
import org.openqa.selenium.By;
import org.openqa.selenium.interactions.Actions;
import org.openqa.selenium.Keys;
import tbc.automation.pages.LoanPage;
import tbc.automation.steps.Landing.clientCenter.GridSteps;
import tbc.automation.utils.Log;

public class LoanSteps {
    private final WindowsDriver winDriver;
    private final LoanPage loanPage;
    private static final Log log = Log.getLogger(LoanSteps.class);

    public LoanSteps(WindowsDriver driver) {
        this.winDriver = driver;
        this.loanPage = new LoanPage(driver);
        log.info("LoanSteps initialized");
    }

    public LoanSteps enterAmount(String amount) {
        log.info("Enter Amount: " + amount);
        loanPage.getAmountInput().clear();
        loanPage.getAmountInput().sendKeys(amount);
        return this;
    }

    public LoanSteps enterPassword(String password) {
        log.info("Enter Password");
        loanPage.getPasswordInput().clear();
        loanPage.getPasswordInput().sendKeys(password);
        return this;
    }

    public LoanSteps selectCurrency(String currency) {
        log.info("Select Currency: " + currency);
        loanPage.getCurrencyDropdown().sendKeys(currency);
        loanPage.getCurrencyDropdown().sendKeys(Keys.ENTER);
        return this;
    }

    public LoanSteps checkInsuredCheckbox() {
        log.info("Check Insured");
        if (!loanPage.getInsuredCheckbox().isSelected()) {
            loanPage.getInsuredCheckbox().click();
        }
        return this;
    }

    public LoanSteps clickShowMessageButton() {
        log.info("Click Show message");
        loanPage.getShowMessageButton().click();
        return this;
    }

    public LoanSteps clickLoansGridCell(int rowIndex, String columnHeader) {
        log.info("Click a cell of loansGrid: row " + rowIndex + ", column " + columnHeader);
        loanPage.getLoansGrid().findElement(By.xpath(".//*[@Name='" + columnHeader + " Row " + rowIndex + "' or starts-with(@Name,'" + columnHeader + " Row " + rowIndex + ",')]")).click();
        return this;
    }

    public LoanSteps doubleClickLoansGridCell(int rowIndex, String columnHeader) {
        log.info("Double-click a cell of loansGrid: row " + rowIndex + ", column " + columnHeader);
        new Actions(winDriver).doubleClick(loanPage.getLoansGrid().findElement(By.xpath(".//*[@Name='" + columnHeader + " Row " + rowIndex + "' or starts-with(@Name,'" + columnHeader + " Row " + rowIndex + ",')]"))).perform();
        return this;
    }

    public LoanSteps rightClickLoansGridCell(int rowIndex, int columnIndex) {
        log.info("Right-click a cell of loansGrid: row " + rowIndex + ", column " + columnIndex);
        new Actions(winDriver).contextClick(loanPage.getLoansGrid().findElement(By.xpath(".//*[@Name='" + " Row " + rowIndex + "' or starts-with(@Name,'" + " Row " + rowIndex + ",')]"))).perform();
        return this;
    }

    public LoanSteps clickGridCell(int rowIndex, String columnHeader) {
        log.info("Click a cell of grid: row " + rowIndex + ", column " + columnHeader);
        new GridSteps(winDriver).clickCell(rowIndex, columnHeader);
        return this;
    }

    public LoanSteps doubleClickGridCell(int rowIndex, String columnHeader) {
        log.info("Double-click a cell of grid: row " + rowIndex + ", column " + columnHeader);
        new GridSteps(winDriver).clickCell(rowIndex, columnHeader);
        new Actions(winDriver).doubleClick().perform();
        return this;
    }

    public LoanSteps clickGridCell2(int rowIndex, String columnHeader) {
        log.info("Click a cell of grid: row " + rowIndex + ", column " + columnHeader);
        loanPage.getGrid().findElement(By.xpath(".//*[@Name='" + columnHeader + " Row " + rowIndex + "' or starts-with(@Name,'" + columnHeader + " Row " + rowIndex + ",')]")).click();
        return this;
    }

    public LoanSteps doubleClickAmountInput() {
        log.info("Double-click Amount");
        new Actions(winDriver).doubleClick(loanPage.getAmountInput()).perform();
        return this;
    }

    public LoanSteps rightClickAmountInput() {
        log.info("Right-click Amount");
        new Actions(winDriver).contextClick(loanPage.getAmountInput()).perform();
        return this;
    }

    public LoanSteps clearAmount() {
        log.info("Clear Amount");
        loanPage.getAmountInput().clear();
        return this;
    }

    public LoanSteps uncheckInsuredCheckbox() {
        log.info("Uncheck Insured");
        if (loanPage.getInsuredCheckbox().isSelected()) {
            loanPage.getInsuredCheckbox().click();
        }
        return this;
    }

    public LoanSteps clickInsuredCheckbox() {
        log.info("Click Insured");
        loanPage.getInsuredCheckbox().click();
        return this;
    }

    public LoanSteps selectFixedRadio() {
        log.info("Select Fixed rate");
        if (!loanPage.getFixedRadio().isSelected()) {
            loanPage.getFixedRadio().click();
        }
        return this;
    }

    public LoanSteps moveToScheduleTab() {
        log.info("Open the Schedule tab");
        loanPage.getScheduleTab().click();
        return this;
    }

    public LoanSteps pressEnterInPassword() {
        log.info("Press Enter in Password");
        loanPage.getPasswordInput().sendKeys(Keys.ENTER);
        return this;
    }

    public LoanSteps clickSaveButton() {
        log.info("Click Save");
        loanPage.getSaveButton().click();
        return this;
    }
}

package tbc.automation.tests;

import org.openqa.selenium.interactions.Actions;
import org.openqa.selenium.Keys;
import org.testng.annotations.Test;
import org.testng.Assert;
import tbc.automation.BaseTest;
import tbc.automation.config.ConfigReader;
import tbc.automation.pages.LoanPage;
import tbc.automation.steps.LoanSteps;
import tbc.automation.steps.MainSteps;
import tbc.automation.steps.WindowSteps;
import tbc.automation.utils.Log;
import tbc.automation.utils.Utility;

@Test(description = "Every step kind")
public class EveryStepKindTest extends BaseTest {

    private static final Log log = Log.getLogger(EveryStepKindTest.class);

    @Test(priority = 1)
    public void everyStepKind() {
        log.info("Every step kind");
        LoanSteps loanSteps = new LoanSteps(winDriver);
        loanSteps
                .enterAmount("5000")
                .enterPassword(ConfigReader.getProperty("passwordInput"))
                .selectCurrency("GEL")
                .checkInsuredCheckbox();
        MainSteps mainSteps = new MainSteps(winDriver);
        mainSteps.clickFileExportPdf();
        loanSteps.clickShowMessageButton();
        WindowSteps windowSteps = new WindowSteps(winDriver);
        windowSteps.clickOkButton();
        loanSteps
                .clickLoansGridCell(2, "Status")
                .doubleClickLoansGridCell(0, "Client")
                .rightClickLoansGridCell(1, 0)
                .clickLoansGridCell(4, "Status")
                .clickGridCell(0, "Client")
                .doubleClickGridCell(0, "Client")
                .clickGridCell2(1, "Select")
                .doubleClickAmountInput()
                .rightClickAmountInput()
                .clearAmount()
                .enterAmount("7 500,00 \"GEL\"")
                .uncheckInsuredCheckbox()
                .clickInsuredCheckbox()
                .selectFixedRadio()
                .moveToScheduleTab()
                .pressEnterInPassword();
        new Actions(winDriver).sendKeys(Keys.chord(Keys.CONTROL, "s")).perform();
        new Utility().printAndSwitchToAllWindows(winDriver);
        LoanPage loanPage = new LoanPage(winDriver);
        Assert.assertTrue(loanPage.getStatusLabel().isDisplayed(), "statusLabel is visible");
        Assert.assertFalse(loanPage.getSaveButton().isEnabled(), "Save is disabled");
        Assert.assertEquals(loanPage.getStatusLabel().getText(), "Ready", "Text of statusLabel");
        Assert.assertEquals(loanPage.getAmountInput().getAttribute("Value.Value"), "7 500,00 \"GEL\"", "Value of Amount");
        Assert.assertFalse(loanPage.getInsuredCheckbox().isSelected(), "Insured is unchecked");
        // Check the totals then save * /
        // Scrolled down 3 notches in loansGrid
        loanSteps.clickSaveButton();
    }
}
